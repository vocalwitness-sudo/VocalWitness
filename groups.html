/** * Direct Join a public group */ export async function joinGroup(groupId) { if (!auth.currentUser) { return showToast('Please sign in to join groups', 'error'); } try { const groupRef = doc(db, 'groups', groupId); const snap = await getDoc(groupRef); if (snap.exists()) { const data = snap.data(); if (data.bannedMembers?.includes(auth.currentUser.uid)) { return showToast('❌ You have been expelled from this group and cannot join.', 'error'); } } await updateDoc(groupRef, { members: arrayUnion(auth.currentUser.uid), memberCount: increment(1) }); showToast('✅ Successfully joined the group!', 'success'); } catch (err) { console.error(err); showToast(err?.message || 'Failed to join group', 'error'); } } /** * Request to Join a restricted group (Approval Queue) */ export async function requestToJoinGroup(groupId) { if (!auth.currentUser) { return showToast('Please sign in to request group access', 'error'); } try { const groupRef = doc(db, 'groups', groupId); const snap = await getDoc(groupRef); if (snap.exists()) { const data = snap.data(); if (data.bannedMembers?.includes(auth.currentUser.uid)) { return showToast('❌ You have been expelled from this group and cannot join.', 'error'); } } await updateDoc(groupRef, { pendingMembers: arrayUnion(auth.currentUser.uid) }); showToast('⏳ Request submitted!
